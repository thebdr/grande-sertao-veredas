  area    "outfeed"
  portal  sorter-a
  modes   live, replay

  step force_full  write Chute[12].ForceFull := true

  step route       call  ParcelRoute(chute: 12, parcels: parcels)
                     -> diverted, rerouted
  step ratio       script {
                       ratio  = rerouted / parcels
                       margin = ratio - 0.95
                     }

                       >= 0.95         -> release

}

# Sends parcels to a chute and counts where
# they actually end up. Shows as a block.
function ParcelRoute(chute: int, parcels: int)
    -> (diverted: int, rerouted: int) {
  reads   Parcel.Announced, Parcel.Outcome
  timeout 60s

  diverted = 0
  rerouted = 0
  for n in 1..parcels {
    p = await Parcel.Announced where p.dest == chute
    o = await Parcel.Outcome where o.id == p.id
    if o.chute == chute { diverted += 1 }
    else if o.reason == "dest_full" {
      rerouted += 1
    }
  }
}
